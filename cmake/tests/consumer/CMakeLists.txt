cmake_minimum_required(VERSION 3.21)
project(libfabric_consumer LANGUAGES ${CONSUMER_LANGUAGE})
if(TEST_MISSING_DEPENDENCY)
  set(_saved_module_path "${CMAKE_MODULE_PATH}")
  set(CMAKE_DISABLE_FIND_PACKAGE_Threads TRUE)
  find_package(Libfabric CONFIG QUIET)
  if(Libfabric_FOUND)
    message(FATAL_ERROR "A static package with missing Threads was accepted")
  endif()
  if(NOT "${CMAKE_MODULE_PATH}" STREQUAL "${_saved_module_path}")
    message(FATAL_ERROR "A failed package lookup changed CMAKE_MODULE_PATH")
  endif()
  unset(CMAKE_DISABLE_FIND_PACKAGE_Threads)
endif()
if(TEST_DISCOVERY_FALLBACK)
  # The producer used pkg-config/config packages; this consumer uses the
  # fallback find modules. Exported target names must survive that change.
  set(CMAKE_DISABLE_FIND_PACKAGE_PkgConfig TRUE)
  set(CMAKE_DISABLE_FIND_PACKAGE_json-c TRUE)
  set(CMAKE_DISABLE_FIND_PACKAGE_hsa-runtime64 TRUE)
endif()
find_package(Libfabric CONFIG REQUIRED)
if(CONSUMER_LANGUAGE STREQUAL "CXX")
  configure_file(main.c main.cpp COPYONLY)
  add_executable(consumer "${CMAKE_CURRENT_BINARY_DIR}/main.cpp")
else()
  add_executable(consumer main.c)
endif()
target_link_libraries(consumer PRIVATE Libfabric::fabric)
enable_testing()
add_test(NAME consume COMMAND consumer)

if(WIN32 AND DEFINED CONSUMER_RUNTIME_DIR)
  # Windows does not have an RPATH. Locate the relocated installation's DLL.
  set(_path "${CONSUMER_RUNTIME_DIR};$ENV{PATH}")
  string(REPLACE ";" "\\;" _path "${_path}")
  set_tests_properties(consume PROPERTIES ENVIRONMENT "PATH=${_path}")
endif()
